% ==================================================
% COMPANY / ORGANISATION LINE
% ==================================================
% Spacing note:
%   The value in \\vspace below controls the gap between the
%   company/organisation line (e.g. "Cybex GmbH") and the first bullet.
%   Increase it for more space; decrease it for a tighter layout.
%   Recommended range: 0.10em to 0.35em.
\newcommand{\cvsubline}[1]{%
  \fontsize{11}{12.8}\selectfont
  \noindent{\color{linkblue}\itshape\bfseries #1}\par
  \vspace{0.22em}%  % <-- GAP: company line -> first bullet
}
